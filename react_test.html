📖 React集成说明

在你的React应用中，使用提供的组件和Hook：

// 1. 导入工具函数
import { checkAuthFromURL, getUserInfo, startWechatAuth } from './utils/wechatAuth';

// 2. 在组件中检查授权状态
useEffect(() => {
  // 检查URL参数中的授权信息
  const authResult = checkAuthFromURL();
  if (authResult.success) {
    setUserInfo(authResult.userInfo);
  } else {
    // 检查本地存储
    const localUserInfo = getUserInfo();
    if (localUserInfo) {
      setUserInfo(localUserInfo);
    }
  }
}, []);

// 3. 发起授权
const handleLogin = () => {
  startWechatAuth();
};
